type document
  define parent: [folder]
  define owner: [user] or owner from parent
  define editor: [user, group#member,
      user with not_expired] or owner
      or editor from parent
  define blocked: [user, group#member]
      or blocked from parent
  define org_editor: editor and org_member
  define sharer: owner or org_editor
  define can_view: viewer but not blocked
  define can_share: sharer but not blocked

Đây là phần chính của model cho type document. Mỗi tính năng của Drive là một dòng DSL. Kế thừa từ thư mục là from parent. Nhóm lồng nhau là userset group thăng member. Public link là wildcard user sao. Chặn thắng mọi quyền là but not blocked, và blocked cũng kế thừa từ thư mục cha nên chặn ở thư mục là chặn cả cây. Khách ngoài org không chia sẻ tiếp được nhờ intersection: sharer là owner, hoặc vừa là editor vừa là member của org. Chia sẻ có hạn là tuple gắn condition not_expired. Admin org có quyền owner qua admin from org ở thư mục, nên tenant boundary nằm ngay trong đồ thị: không có tuple nào nối sang org khác thì không có đường quyền.
